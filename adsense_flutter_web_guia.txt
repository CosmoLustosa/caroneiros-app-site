  import 'dart:html' as html;
  import 'package:flutter/foundation.dart';
  import 'package:flutter/material.dart';

  class AdSenseBanner extends StatefulWidget {
    final String adClient;
    final String adSlot;
    final String adFormat;
    final double height;
    final EdgeInsetsGeometry margin;

    const AdSenseBanner({
      super.key,
      required this.adClient,
      required this.adSlot,
      this.adFormat = 'auto',
      this.height = 90,
      this.margin = const EdgeInsets.symmetric(vertical: 12),
    });

    @override
    State<AdSenseBanner> createState() => _AdSenseBannerState();
  }

  class _AdSenseBannerState extends State<AdSenseBanner> {
    static bool _scriptLoaded = false;

    @override
    void initState() {
      super.initState();
      if (kIsWeb) _loadAdSenseScript();
    }

    void _loadAdSenseScript() {
      if (_scriptLoaded) { _pushAd(); return; }
      final script = html.ScriptElement()
        ..src = 'https://pagead2.googlesyndication.com/pagead/js/adsbygoogle.js?client=${widget.adClient}'
        ..async = true
        ..crossOrigin = 'anonymous';
      html.document.head!.append(script);
      _scriptLoaded = true;
      script.onLoad.listen((_) => _pushAd());
    }

    void _pushAd() {
      try {
        (html.window as dynamic).adsbygoogle.push({});
      } catch (_) {}
    }

    @override
    Widget build(BuildContext context) {
      if (!kIsWeb) return const SizedBox.shrink();
      return Container(
        margin: widget.margin,
        width: double.infinity,
        child: AdWidget(content: _buildAdHtml()),
      );
    }

    html.Element _buildAdHtml() {
      final div = html.DivElement();
      final ins = html.InsElement()
        ..className = 'adsbygoogle'
        ..style.display = 'block'
        ..style.height = '${widget.height}px'
        ..dataset['adClient'] = widget.adClient
        ..dataset['adSlot'] = widget.adSlot
        ..dataset['adFormat'] = widget.adFormat
        ..dataset['fullWidthResponsive'] = 'true';
      div.append(ins);
      return div;
    }
  }

  class AdWidget extends StatelessWidget {
    final html.Element content;
    const AdWidget({super.key, required this.content});

    @override
    Widget build(BuildContext context) {
      // Usando HtmlElementView para renderizar HTML nativo no Flutter Web
      return HtmlElementView(viewType: 'ad-${content.hashCode}');
    }
  }
